#include <stdlib.h>
#include <string.h>
void backtrack(int n, int openCount, int closeCount, char* currentString, int index, 
               char*** result, int* capacity, int* returnSize) {
if (openCount == n && closeCount == n) {
        currentString[index] = '\0'; // Null-terminate the string
        if (*returnSize >= *capacity) {
            *capacity *= 2;
            *result = (char**)realloc(*result, (*capacity) * sizeof(char*));
        }
        (*result)[*returnSize] = (char*)malloc((2 * n + 1) * sizeof(char));
        strcpy((*result)[*returnSize], currentString);
        (*returnSize)++;
        return;
    }
    if (openCount < n) {
        currentString[index] = '(';
        backtrack(n, openCount + 1, closeCount, currentString, index + 1, result, capacity, returnSize);
    }
    if (closeCount < openCount) {
        currentString[index] = ')';
        backtrack(n, openCount, closeCount + 1, currentString, index + 1, result, capacity, returnSize);
    }
}
 generateParenthesis(int n, int* returnSize) {
    *returnSize = 0;
    if (n <= 0) return NULL;
    int capacity = 50; 
    char** result = (char**)malloc(capacity * sizeof(char*));
    char* currentString = (char*)malloc((2 * n + 1) * sizeof(char);
    backtrack(n, 0, 0, currentString, 0, &result, &capacity, returnSize);
    free(currentString);
    return result;
}
